(function () { const AD_JSON_PATH = "ad.json"; const TOKEN_KEY = "sehati_gh_token"; let adsList = []; let currentAdIdx = 0; let defaultDurationSec = 5; let adTimer = null; function cleanPromoLabels() { document.querySelectorAll("*").forEach((el) => { if (el.children.length === 0) { const txt = (el.textContent || "").trim(); if (txt.includes("مساحة إعلانية مميزة")) { el.style.display = "none"; } } }); } function findAdContainer() { return ( document.getElementById("sehatiAdBox") || document.querySelector(".ad-banner") || document.querySelector(".ad-container") || document.querySelector(".ad-card") ); } function renderCurrentAd() { cleanPromoLabels(); const box = findAdContainer(); if (!box) return; if (!adsList.length) { box.innerHTML = ""; box.style.display = "none"; return; } const ad = adsList[currentAdIdx % adsList.length]; box.style.display = "block"; box.style.opacity = "1"; box.style.filter = "none"; const title = (ad.title || "").trim(); const desc = (ad.description || "").trim(); const img = (ad.image || "").trim(); const btnText = (ad.buttonText || "تواصل الآن").trim(); const btnLink = (ad.buttonLink || "#").trim(); box.innerHTML = `
${ img ? `` : "" }
${ title ? `
${title}
` : "" } ${ desc ? `
${desc}
` : "" } ${ btnLink && btnLink !== "#" ? `${btnText}` : "" }
`; } function scheduleNextAd() { if (adTimer) clearTimeout(adTimer); if (adsList.length <= 1) return; const currentAd = adsList[currentAdIdx % adsList.length]; const sec = Math.max(1, Number(currentAd.durationSec || defaultDurationSec || 5)); adTimer = setTimeout(() => { currentAdIdx = (currentAdIdx + 1) % adsList.length; renderCurrentAd(); scheduleNextAd(); }, sec * 1000); } async function loadSehatiAds() { try { const res = await fetch(`${AD_JSON_PATH}?v=${Date.now()}`, { cache: "no-store" }); if (res.ok) { const data = await res.json(); if (data && data.defaultDurationSec) { defaultDurationSec = Math.max(1, Number(data.defaultDurationSec) || 5); } if (Array.isArray(data.ads)) { adsList = data.ads; } else if (data && data.active !== false && (data.title || data.description || data.image)) { adsList = [ { id: "legacy_ad", title: data.title || "", description: data.description || "", image: data.image || "", buttonText: data.buttonText || "تواصل الآن", buttonLink: data.buttonLink || "#", durationSec: defaultDurationSec } ]; } else { adsList = []; } } } catch (_) {} currentAdIdx = 0; renderCurrentAd(); scheduleNextAd(); } window.addEventListener("DOMContentLoaded", () => { cleanPromoLabels(); loadSehatiAds(); }); })();
